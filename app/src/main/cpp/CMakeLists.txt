cmake_minimum_required(VERSION 3.22.1)
project(audiophile_player LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_library(log-lib log)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/peq_engine.cpp" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/peq_jni.cpp")
 add_library(peq SHARED peq_engine.cpp peq_jni.cpp)
 target_compile_features(peq PRIVATE cxx_std_17)
 target_compile_options(peq PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-fast-math)
 target_link_options(peq PRIVATE "-Wl,-z,max-page-size=16384")
 target_include_directories(peq PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(peq PRIVATE ${log-lib})
endif()

add_library(audiophile_immersion SHARED ImmersionEngine.cpp ImmersionNative.cpp)
target_compile_features(audiophile_immersion PRIVATE cxx_std_17)
target_compile_options(audiophile_immersion PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-fast-math)
target_link_options(audiophile_immersion PRIVATE "-Wl,-z,max-page-size=16384")
target_include_directories(audiophile_immersion PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(audiophile_immersion PRIVATE ${log-lib})

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native_vocal_remover.cpp")
 if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ProfessionalStereoWidenerDSP_v10.h")
  message(FATAL_ERROR "ProfessionalStereoWidenerDSP_v10.h is required by the existing vocal-remover DSP dependency.")
 endif()
 add_library(audiophile_vocal_remover SHARED native_vocal_remover.cpp)
 target_compile_features(audiophile_vocal_remover PRIVATE cxx_std_17)
 target_compile_options(audiophile_vocal_remover PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-fast-math)
 target_link_options(audiophile_vocal_remover PRIVATE "-Wl,-z,max-page-size=16384")
 target_include_directories(audiophile_vocal_remover PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(audiophile_vocal_remover PRIVATE ${log-lib})
endif()
